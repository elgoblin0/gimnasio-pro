"""capturar.py SALIDA [pantallas...]: capturas de chequeo, resumen y editor."""
import json, sys, pathlib
from playwright.sync_api import sync_playwright
BASE="http://127.0.0.1:5003"
sal=pathlib.Path(sys.argv[1]); sal.mkdir(parents=True,exist_ok=True)
pedido=set(sys.argv[2:])
CON="try{localStorage.setItem('gimnasio-sin-efectos','no')}catch{}"
SIN="try{localStorage.setItem('gimnasio-sin-efectos','si')}catch{}"
RESUMEN=dict(zonas_trabajadas=["pecho","hombros","triceps","cuadriceps","gluteos","abdomen"],zonas_con_dolor=["rodillas"],records=[],sustituciones=[],pendientes=[],ejercicios=[])
def abrir(b,ancho,alto,esquema="light",init=CON,boton="Entrar como socio",tacto=False):
    c=b.new_context(viewport=dict(width=ancho,height=alto),locale="es-PY",color_scheme=esquema,service_workers="block",has_touch=tacto,device_scale_factor=2 if tacto else 1)
    c.add_init_script(init); p=c.new_page(); p.goto(BASE); p.get_by_role("button",name=boton,exact=True).click(); p.wait_for_timeout(600)
    return c,p
def esperar(p,sel):
    try: p.locator(sel+'[data-cuerpo3d="listo"]').first.wait_for(timeout=30000)
    except Exception as e: print("sin 3D",sel,str(e)[:80])
    p.wait_for_timeout(1400)
def chequeo(b,nom,**k):
    c,p=abrir(b,**k); p.goto(BASE+"/entrenamiento.html"); p.wait_for_load_state("load")
    return c,p
def marcar(p,zs):
    for z in zs:
        v=p.get_by_role("button",name="Ver todas las zonas",exact=True).first
        if v.count() and v.is_visible(): v.click()
        p.get_by_role("checkbox",name=z,exact=True).first.click(); p.wait_for_timeout(80)
    p.wait_for_timeout(1000)
def foto(p,n,sel=None):
    if sel:
        l=p.locator(sel).first; l.scroll_into_view_if_needed(); p.wait_for_timeout(300); l.screenshot(path=str(sal/f"{n}.png"))
    else: p.screenshot(path=str(sal/f"{n}.png"))
    print(n)
def resumen(b,**k):
    c,p=abrir(b,**k); p.route("**/api/entrenamiento/sesiones/*/resumen",lambda r:r.fulfill(json=RESUMEN)); p.goto(BASE+"/resumen-sesion.html?sesion=demo"); return c,p
def editor(b,**k):
    k.setdefault("boton","Entrar como entrenador"); c,p=abrir(b,**k); p.wait_for_timeout(600)
    socios=p.evaluate("fetch('/api/sesion/demo/socios').then(r=>r.json())")["socios"]; s=socios[0]["id"]
    p.goto(f"{BASE}/editor-programa.html?socio={s}"); return c,p
hay=lambda x: not pedido or x in pedido
with sync_playwright() as w:
    b=w.chromium.launch(args=["--use-angle=d3d11","--enable-gpu","--ignore-gpu-blocklist"])
    if hay("chequeo"):
        for nom,k in [("chequeo-escritorio-dia",dict(ancho=1440,alto=900)),("chequeo-escritorio-noche",dict(ancho=1440,alto=900,esquema="dark")),("chequeo-celular-dia",dict(ancho=390,alto=844,tacto=True))]:
            c,p=chequeo(b,nom,**k); esperar(p,"[data-zonas-chequeo]"); foto(p,nom+"-vacio","[data-zonas-chequeo]")
            marcar(p,["Pecho","Hombros","Rodillas"]); foto(p,nom+"-marcado","[data-zonas-chequeo]")
            b2=p.get_by_role("button",name="Ver de espalda").first
            if b2.count(): b2.click(); p.wait_for_timeout(900); foto(p,nom+"-espalda","[data-zonas-chequeo]")
            c.close()
    if hay("resumen"):
        for nom,k in [("resumen-escritorio-dia",dict(ancho=1440,alto=900)),("resumen-celular-noche",dict(ancho=390,alto=844,esquema="dark",tacto=True))]:
            c,p=resumen(b,**k); esperar(p,"[data-zonas-resumen]"); foto(p,nom); c.close()
    if hay("editor"):
        for nom,k in [("editor-escritorio-dia",dict(ancho=1440,alto=900)),("editor-celular-dia",dict(ancho=390,alto=844,tacto=True))]:
            c,p=editor(b,**k); esperar(p,"[data-volumen]"); foto(p,nom); foto(p,nom+"-cuerpo",".editor-cuerpo"); c.close()
    if hay("alternativa"):
        c,p=chequeo(b,"sin",ancho=1440,alto=900,init=SIN); p.wait_for_timeout(1200); foto(p,"sin-efectos-chequeo","[data-zonas-chequeo]"); c.close()
        c,p=resumen(b,ancho=1440,alto=900,init=SIN); p.wait_for_timeout(1200); foto(p,"sin-efectos-resumen"); c.close()
        c,p=editor(b,ancho=1440,alto=900,init=SIN); p.wait_for_timeout(1200); foto(p,"sin-efectos-editor",".editor-cuerpo"); c.close()
    if hay("carga"):
        c,p=abrir(b,1440,900); p.route("**/vendor/cuerpo*/*",lambda r:None); p.goto(BASE+"/entrenamiento.html"); p.wait_for_timeout(2500); foto(p,"carga-chequeo","[data-zonas-chequeo]"); c.close()
    if hay("figura"):
        c,p=chequeo(b,"f",ancho=1440,alto=900); esperar(p,"[data-zonas-chequeo]")
        for nombre in ["Masculina","Femenina"]:
            btn=p.get_by_role("button",name=nombre,exact=True).first
            btn.click(); p.wait_for_timeout(1800); foto(p,"figura-"+nombre.lower(),"[data-zonas-chequeo]")
        c.close()
    b.close()
