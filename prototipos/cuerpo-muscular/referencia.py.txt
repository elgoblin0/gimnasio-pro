# Proyecta cada figura, en la postura de la cruz, a una imagen con grilla para dibujar los músculos encima.
import bpy, sys, json, math
import numpy as np
sal = sys.argv[sys.argv.index("--")+1]
ALTURA=1.79
F={"GEO-body_male_realistic":("m",dict(pivote=(.2,1.42),caida=1.19,axila=(.175,1.27,.397))),
   "GEO-body_female_realistic":("f",dict(pivote=(.16,1.42),caida=1.18,axila=(.155,1.27,.405)))}
def suave(t): return t*t*(3-2*t)
def enderezar(p,cfg):
    (px,py),c,(ax,ay,ap)=cfg["pivote"],cfg["caida"],cfg["axila"]
    x,y,z=p[:,0],p[:,1],p[:,2]
    linea=ax+np.maximum(ay-y,0)*ap
    t=np.clip((np.abs(x)-linea)/.04,0,1); a=c*suave(t)
    dx=np.abs(x)-px; dy=y-py; s,co=np.sin(a),np.cos(a)
    nx=np.where(t>0,np.sign(x)*(px+dx*co-dy*s),x); ny=np.where(t>0,py+dx*s+dy*co,y)
    return np.stack([nx,ny,z],1)
for nombre,(et,cfg) in F.items():
    o=bpy.data.objects[nombre]
    for m in o.modifiers:
        if m.type=="MULTIRES": m.levels=1
    me=bpy.data.meshes.new_from_object(o.evaluated_get(bpy.context.evaluated_depsgraph_get())); me.transform(o.matrix_world)
    v=np.array([x.co[:] for x in me.vertices]); 
    v[:,2]-=v[:,2].min(); v*=ALTURA/v[:,2].max(); v[:,0]-=(v[:,0].min()+v[:,0].max())/2
    # glTF: x, z, -y  → en la app y arriba y z hacia el frente
    p=np.stack([v[:,0],v[:,2],-v[:,1]],1)
    r=p
    np.save(f"{sal}/ref/{et}_rep.npy",r)
    me.calc_loop_triangles()
    tri=np.array([t.vertices[:] for t in me.loop_triangles]); np.save(f"{sal}/ref/{et}_tri.npy",tri)
    n=np.array([x.normal[:] for x in me.vertices]); np.save(f"{sal}/ref/{et}_nor.npy",np.stack([n[:,0],n[:,2],-n[:,1]],1))
    print("LISTO",et,len(v),r.min(0),r.max(0))
