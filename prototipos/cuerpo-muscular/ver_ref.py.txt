import sys,pathlib,json
from playwright.sync_api import sync_playwright
# x en metros → px: (x+.02)*1000; y → 1860-y*1000
d=pathlib.Path(sys.argv[1]).as_posix(); fig=sys.argv[2] if len(sys.argv)>2 else "mf"
def lam(y0,y1):
    out=""
    for i,lado in enumerate(("frente","espalda")):
        g=""
        for k in range(0,13):
            x=(k*.05+.02)*1000; g+=f'<line x1="{x}" y1="0" x2="{x}" y2="1860" stroke="{"#d00" if k%2==0 else "#f99"}" stroke-width="{1 if k%2==0 else .6}"/>'
            if k%2==0: g+=f'<text x="{x+2}" y="{1860-y1*1000+14}" font-size="13" fill="#c00">{k*.05:.1f}</text>'
        for k in range(0,38):
            y=1860-k*50; g+=f'<line x1="0" y1="{y}" x2="620" y2="{y}" stroke="{"#06c" if k%2==0 else "#7bf"}" stroke-width="{1 if k%2==0 else .6}"/>'
            g+=f'<text x="2" y="{y-2}" font-size="13" fill="#06c">{k*.05:.2f}</text>'
        out+=f'<svg width="620" height="{(y1-y0)*1000}" viewBox="0 {1860-y1*1000} 620 {(y1-y0)*1000}" style="margin-right:6px"><image href="file:///{d}/{et}_{lado}.png" width="620" height="1860"/>{g}</svg>'
    return f'<body style="margin:0;display:flex">{out}</body>'
with sync_playwright() as w:
    b=w.chromium.launch()
    for et in fig:
      for parte,y0,y1 in (("alto",.9,1.82),("bajo",0,.95)):
        p=b.new_page(viewport=dict(width=1250,height=int((y1-y0)*1000)))
        open(f"{d}/t.html","w",encoding="utf-8").write(lam(y0,y1)); p.goto(f"file:///{d}/t.html"); p.wait_for_timeout(300); p.screenshot(path=f"{d}/g_{et}_{parte}.png"); p.close()
    b.close()
