"""Demo local para diseñar: base propia, semilla fresca y servidor en 5003."""
import os, sys
sys.path.insert(0, os.getcwd())
os.environ.update(GIMNASIO_BD_HOST="127.0.0.1", GIMNASIO_BD_PUERTO="3306", GIMNASIO_BD_USUARIO="root",
    GIMNASIO_BD_CLAVE="solo-ci", GIMNASIO_BD_NOMBRE="gimnasio_cuerpo", MODO_DEMO="1", GIMNASIO_CLAVE_SECRETA="solo-diseno-local")
from gimnasio import crear_app
from gimnasio.comun.bd import conectar, parametros_desde_entorno
from gimnasio.comun.migrar import aplicar
from gimnasio.demo.servicio import reiniciar
with conectar(parametros_desde_entorno() | {"database": None}) as c, c.cursor() as cur:
    cur.execute("DROP DATABASE IF EXISTS gimnasio_cuerpo")
    cur.execute("CREATE DATABASE gimnasio_cuerpo CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci")
aplicar(parametros_desde_entorno(), salida=lambda _: None)
app = crear_app()
with app.app_context():
    reiniciar()
app.extensions["limite_demo"].espera = lambda ip: 0
app.run(port=5003)
